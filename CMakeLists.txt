cmake_minimum_required(VERSION 3.16)
project(iot_stream_framework
    VERSION 1.0.0
    DESCRIPTION "Reactive Object-Oriented Stream Processing Pipeline Framework in C++17"
    LANGUAGES CXX
)

# ---------------------------------------------------------------------------
# C++17 Standard Settings
# ---------------------------------------------------------------------------
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Export compile_commands.json cho các công cụ LSP (clangd, ccls, VS Code)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# ---------------------------------------------------------------------------
# Compiler Flags: Warnings & Tối ưu hóa -O2 cho Release
# ---------------------------------------------------------------------------
if(MSVC)
    add_compile_options(/W4 /permissive- /EHsc)
    add_compile_options($<$<CONFIG:Release>:/O2>)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
    add_compile_options($<$<CONFIG:Release>:-O2>)
endif()

# ---------------------------------------------------------------------------
# Dependencies: Threads (Zero External Libraries)
# ---------------------------------------------------------------------------
find_package(Threads REQUIRED)

# ---------------------------------------------------------------------------
# Target Library: iot_stream (STATIC)
# Gom tất cả src/**/*.cpp ngoại trừ main.cpp
# ---------------------------------------------------------------------------
file(GLOB_RECURSE IOT_STREAM_SOURCES 
    CONFIGURE_DEPENDS 
    "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp"
)
list(FILTER IOT_STREAM_SOURCES EXCLUDE REGEX ".*/main\\.cpp$")

# Đảm bảo target STATIC luôn hợp lệ ngay cả khi chưa tạo các file .cpp khác
if(NOT IOT_STREAM_SOURCES)
    set(STUB_FILE "${CMAKE_CURRENT_BINARY_DIR}/iot_stream_stub.cpp")
    file(WRITE "${STUB_FILE}" "// Autogenerated stub for empty static library\nnamespace iot_stream { void _init_stub() {} }\n")
    set(IOT_STREAM_SOURCES "${STUB_FILE}")
endif()

add_library(iot_stream STATIC ${IOT_STREAM_SOURCES})

target_include_directories(iot_stream PUBLIC 
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>
)

target_link_libraries(iot_stream PUBLIC Threads::Threads)

# ---------------------------------------------------------------------------
# Target Executable: iot_app (từ src/main.cpp)
# ---------------------------------------------------------------------------
add_executable(iot_app src/main.cpp)
target_include_directories(iot_app PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(iot_app PRIVATE iot_stream Threads::Threads)

# ---------------------------------------------------------------------------
# Subdirectories: tests và benchmark
# ---------------------------------------------------------------------------
enable_testing()
add_subdirectory(tests)
add_subdirectory(benchmark)
